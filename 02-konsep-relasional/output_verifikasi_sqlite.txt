A. Latihan 2 & 3: logika NULL (tiga nilai)
==============================================================================
WHERE email = NULL     -> 0 baris
WHERE email IS NULL    -> 2 baris
WHERE email <> 'a@x.com' -> 1 baris (baris NULL ikut tersaring)
WHERE email IS DISTINCT FROM 'a@x.com' (SQLite: IS NOT) -> 3 baris

NULL OR TRUE                                            -> 1   (None = NULL, 1 = true, 0 = false)
NULL AND FALSE                                          -> 0   (None = NULL, 1 = true, 0 = false)
COALESCE(NULL, NULL, 3)                                 -> 3   (None = NULL, 1 = true, 0 = false)
NULL IS DISTINCT FROM NULL (SQLite: NULL IS NOT NULL)   -> 0   (None = NULL, 1 = true, 0 = false)
NULL = NULL                                             -> None   (None = NULL, 1 = true, 0 = false)
TRUE AND NULL                                           -> None   (None = NULL, 1 = true, 0 = false)
FALSE AND NULL                                          -> 0   (None = NULL, 1 = true, 0 = false)
5 + NULL                                                -> None   (None = NULL, 1 = true, 0 = false)
